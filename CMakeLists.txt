cmake_minimum_required(VERSION 3.16)
project(verb VERSION 0.1.16 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Threads REQUIRED)

add_library(verbcore STATIC src/render.cpp src/tail.cpp src/rt_engine.cpp src/energy.cpp)
set_target_properties(verbcore PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(verbcore PUBLIC Threads::Threads)
target_include_directories(verbcore PUBLIC src)
if(MSVC)
  target_compile_options(verbcore PRIVATE /utf-8)
endif()
set(VERB_X86_TARGET OFF)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|x64|i[3-6]86)$")
  set(VERB_X86_TARGET ON)
endif()
# A universal Apple build must use flags supported by every slice, including arm64.
if(APPLE AND CMAKE_OSX_ARCHITECTURES MATCHES "(^|;)arm64($|;)")
  set(VERB_X86_TARGET OFF)
endif()
option(VERB_AVX2 "Require AVX2/FMA for the engine (x86 targets only)" ${VERB_X86_TARGET})
if(VERB_AVX2 AND NOT VERB_X86_TARGET)
  message(FATAL_ERROR "VERB_AVX2 requires an x86-only target. Use -DVERB_AVX2=OFF for ARM or universal builds.")
endif()
target_compile_definitions(verbcore PUBLIC VERB_REQUIRE_AVX2=$<BOOL:${VERB_AVX2}>)
if(NOT MSVC)
  target_compile_options(verbcore PRIVATE -O3 -Wall -Wextra)
  if(VERB_AVX2)
    target_compile_options(verbcore PRIVATE -mavx2 -mfma)
  endif()
elseif(VERB_AVX2)
  target_compile_options(verbcore PRIVATE /arch:AVX2)
endif()

add_executable(verb src/main.cpp)
target_link_libraries(verb PRIVATE verbcore)

add_executable(rt_test src/rt_test.cpp)
target_link_libraries(rt_test PRIVATE verbcore)

add_executable(tail_regression src/tail_regression.cpp)
target_link_libraries(tail_regression PRIVATE verbcore)
enable_testing()
add_test(NAME tail_regression COMMAND tail_regression)
add_test(NAME cathedral_tail COMMAND tail_regression)
set_tests_properties(cathedral_tail PROPERTIES ENVIRONMENT "VERB_CATHEDRAL_ONLY=1")
add_test(NAME vsl_hall_tail COMMAND tail_regression)
set_tests_properties(vsl_hall_tail PROPERTIES ENVIRONMENT "VERB_VSL_TAIL_ONLY=1")
add_executable(quality_test src/quality_test.cpp)
target_link_libraries(quality_test PRIVATE verbcore)
add_test(NAME quality_test COMMAND quality_test)
add_executable(energy_test src/energy_test.cpp)
target_link_libraries(energy_test PRIVATE verbcore)
add_test(NAME energy_map COMMAND energy_test)
set_tests_properties(quality_test PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")

# JUCE plugin (needs MSVC on Windows): cmake -DVERB_BUILD_PLUGIN=ON
option(VERB_BUILD_PLUGIN "Build the JUCE VST3 plugin" OFF)
if(VERB_BUILD_PLUGIN)
  add_subdirectory(third_party/JUCE)
  add_subdirectory(plugin)
endif()

# E1 objective focused tests (not a long-suite invocation).
add_executable(e1_wet_test src/e1_wet_test.cpp)
target_link_libraries(e1_wet_test PRIVATE verbcore)
add_test(NAME e1_wet COMMAND e1_wet_test)
set_tests_properties(e1_wet PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 180)

# E2 listening renderer (tool only, not a test; renders out/E2/listen).
add_executable(e2_listen src/e2_listen.cpp)
target_link_libraries(e2_listen PRIVATE verbcore)
add_executable(e2_bench src/e2_bench.cpp)
target_link_libraries(e2_bench PRIVATE verbcore)
target_compile_definitions(e2_bench PRIVATE E2_NEW=1)
add_executable(e2_freeze_test src/e2_freeze_test.cpp)
target_link_libraries(e2_freeze_test PRIVATE verbcore)
add_test(NAME e2_freeze COMMAND e2_freeze_test)
set_tests_properties(e2_freeze PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 600)
