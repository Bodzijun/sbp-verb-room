cmake_minimum_required(VERSION 3.16)
project(verb VERSION 0.1.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Threads REQUIRED)

add_library(verbcore STATIC src/render.cpp src/tail.cpp src/rt_engine.cpp)
target_link_libraries(verbcore PUBLIC Threads::Threads)
target_include_directories(verbcore PUBLIC src)
if(MSVC)
  target_compile_options(verbcore PRIVATE /utf-8)
endif()
option(VERB_AVX2 "Compile the engine with AVX2/FMA (2.4x faster; the plugin refuses to run engine code on CPUs without it)" ON)
if(NOT MSVC)
  target_compile_options(verbcore PRIVATE -O3 -Wall -Wextra)
  if(VERB_AVX2)
    target_compile_options(verbcore PRIVATE -mavx2 -mfma)
  endif()
elseif(VERB_AVX2)
  target_compile_options(verbcore PRIVATE /arch:AVX2)
endif()

add_executable(verb src/main.cpp)
target_link_libraries(verb PRIVATE verbcore)

add_executable(rt_test src/rt_test.cpp)
target_link_libraries(rt_test PRIVATE verbcore)

# JUCE plugin (needs MSVC on Windows): cmake -DVERB_BUILD_PLUGIN=ON
option(VERB_BUILD_PLUGIN "Build the JUCE VST3 plugin" OFF)
if(VERB_BUILD_PLUGIN)
  add_subdirectory(third_party/JUCE)
  add_subdirectory(plugin)
endif()
